  defp walk({:case, scrut, _motive, branches}, depth, _site, st) do
    scrut_check =
      if collapsible_case?(st.env, branches),

      each(branches, fn {cname, arity, body} ->
        ctor_qs =

        branch_erased =

          |> Enum.with_index()
          |> Enum.filter(fn {q, _p} -> q == :erased end)
          |> Enum.map(fn {_q, p} -> depth + p end)
